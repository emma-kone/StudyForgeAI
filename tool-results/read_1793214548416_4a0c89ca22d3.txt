   148→function hashStr(s: string): number {
   149→  let h = 2166136261;
   150→  for (let i = 0; i < s.length; i++) {
   151→    h ^= s.charCodeAt(i);
   152→    h = Math.imul(h, 16777619);
   153→  }
   154→  return h >>> 0;
   155→}

   157→/** Mélange déterministe (même graine → même ordre, stable entre rendus). */
   158→function seedShuffle<T>(arr: T[], seed: number): T[] {
   159→  const copy = [...arr];
   160→  let state = seed || 1;
   161→  const rand = () => {
   162→    state |= 0;
   163→    state = (state + 0x6d2b79f5) | 0;
   164→    let t = Math.imul(state ^ (state >>> 15), 1 | state);
   165→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
   166→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   167→  };
   168→  for (let i = copy.length - 1; i > 0; i--) {
   169→    const j = Math.floor(rand() * (i + 1));
   170→    [copy[i], copy[j]] = [copy[j], copy[i]];
   171→  }
   172→  return copy;
   173→}

   200→function useIsDark(): boolean {
   201→  const [isDark, setIsDark] = useState(false);
   202→  useEffect(() => {
   203→    const root = document.documentElement;
   204→    const update = () => setIsDark(root.classList.contains("dark"));
   205→    update();
   206→    const observer = new MutationObserver(update);
   207→    observer.observe(root, { attributes: true, attributeFilter: ["class"] });
   208→    return () => observer.disconnect();
   209→  }, []);
   210→  return isDark;
   211→}

   214→  switch (q.type) {

   240→function hasAnswer(q: Question, draft: unknown): boolean {
   241→  if (draft == null) return false;
   242→  switch (q.type) {
   243→    case "qcm_single":
   244→      return typeof draft === "string" && draft.length > 0;
   245→    case "qcm_multiple":
   246→      return Array.isArray(draft) && draft.length > 0;
   247→    case "true_false":
   248→      return draft === "true" || draft === "false";
   249→    case "fill_blank":
   250→      return Array.isArray(draft) && draft.some((v) => String(v ?? "").trim() !== "");
   251→    case "matching":
   252→      return typeof draft === "object" && Object.values(draft as Record<string, string>).some((v) => v);
   253→    case "ordering":
   254→      return Array.isArray(draft) && draft.length > 0;
   255→    case "error_identification":
   256→      return typeof draft === "object" && String((draft as { line?: string }).line ?? "").trim() !== "";
   257→    case "open_question":
   258→    case "case_study":
   259→    case "data_modeling":
   260→    case "combined":
   261→    case "exam_composite":
   262→    case "short_answer":
   263→    case "code_output":
   264→      return typeof draft === "string" && draft.trim().length > 0;
   265→    case "code_write":
   266→    case "code_fix":
   267→    case "debug":
   268→    case "sql_query":
   269→    case "sql_create_table":
   270→    case "sql_database":
   271→      return typeof draft === "string" && draft.trim().length > 0;
   272→    default:
   273→      return String(draft).trim().length > 0;
   274→  }
   275→}

   430→function FillBlankInput({ q, value, onChange }: { q: Question; value: unknown; onChange: (v: string[]) => void }) {
   431→  const answers = Array.isArray(value) ? (value as string[]) : [];
   432→  const blanks = q.blanks ?? [];
   433→  const segments = q.segments ?? [];
   434→  const set = (i: number, v: string) => {
   435→    const next = [...answers];
   436→    while (next.length < blanks.length) next.push("");
   437→    next[i] = v;
   438→    onChange(next);
   439→  };
   440→  return (
   441→    <div className="space-y-3">
   442→      <div className="flex flex-wrap items-center gap-2 text-sm leading-loose">
   443→        {segments.map((seg, i) => (
   444→          <span key={i} className="contents">
   445→            <span className="whitespace-pre-wrap">{seg}</span>
   446→            {i < blanks.length && (
   447→              <Input
   448→                value={answers[i] ?? ""}
   449→                onChange={(e) => set(i, e.target.value)}
   450→                className="inline-block w-40 font-medium"
   451→                aria-label={`Trou ${i + 1}`}
   452→                autoComplete="off"
   453→              />
   454→            )}
   455→          </span>
   456→        ))}
   457→      </div>
   458→      {blanks.length > segments.length - 1 && (
   459→        <div className="space-y-2">
   460→          {blanks.slice(Math.max(0, segments.length - 1)).map((_, j) => {
   461→            const idx = Math.max(0, segments.length - 1) + j;
   462→            return (
   463→              <Input
   464→                key={idx}
   465→                value={answers[idx] ?? ""}
   466→                onChange={(e) => set(idx, e.target.value)}
   467→                placeholder={`Réponse ${idx + 1}`}
   468→                aria-label={`Trou ${idx + 1}`}
   469→                autoComplete="off"
   470→              />
   471→            );
   472→          })}
   473→        </div>
   474→      )}
   475→      <p className="typo-meta">{blanks.length} trou(s) à compléter. Les variantes orthographiques raisonnables sont acceptées.</p>
   476→    </div>
   477→  );
   478→}

   480→function MatchingInput({
   481→  q,
   482→  value,
   483→  onChange,
   484→}: {
   485→  q: Question;
   486→  value: unknown;
   487→  onChange: (v: Record<string, string>) => void;
   488→}) {
   489→  const pairs = q.pairs ?? [];
   490→  const rights = useMemo(() => seedShuffle(pairs.map((p) => p.right), hashStr(q.statement) || 7), [pairs, q.statement]);
   491→  const current = (typeof value === "object" && value !== null ? value : {}) as Record<string, string>;
   492→  return (
   493→    <div className="space-y-3">
   494→      {pairs.map((p, i) => (
   495→        <div key={i} className="grid gap-2 sm:grid-cols-[1fr_auto_1fr] sm:items-center">
   496→          <span className="rounded-md border bg-muted/40 px-3 py-2 text-sm font-medium">{p.left}</span>
   497→          <ArrowRight className="hidden h-4 w-4 text-muted-foreground sm:block" aria-hidden />
   498→          <Select value={current[p.left] ?? ""} onValueChange={(v) => onChange({ ...current, [p.left]: v })}>
   499→            <SelectTrigger className="w-full sm:w-64" aria-label={`Association pour ${p.left}`}>
   500→              <SelectValue placeholder="Choisir la définition…" />
   501→            </SelectTrigger>
   502→            <SelectContent>
   503→              {rights.map((r, j) => (
   504→                <SelectItem key={`${r}-${j}`} value={r}>
   505→                  {r}
   506→                </SelectItem>
   507→              ))}
   508→            </SelectContent>
   509→          </Select>
   510→        </div>
   511→      ))}
   512→      <p className="typo-meta">Associez chaque notion à la bonne définition.</p>
   513→    </div>
   514→  );

   517→function OrderedListInput({
   518→  q,
   519→  value,
   520→  onChange,
   521→}: {
   522→  q: Question;
   523→  value: unknown;
   524→  onChange: (v: string[]) => void;
   525→}) {
   526→  const items = useMemo(
   527→    () => seedShuffle(q.orderedItems ?? [], hashStr(q.statement) || 11),
   528→    [q.orderedItems, q.statement]
   529→  );
   530→  const list = Array.isArray(value) && value.length > 0 ? (value as string[]) : items;
   531→  const move = (i: number, dir: -1 | 1) => {
   532→    const next = [...list];
   533→    const j = i + dir;
   534→    if (j < 0 || j >= next.length) return;
   535→    [next[i], next[j]] = [next[j], next[i]];
   536→    onChange(next);
   537→  };
   538→  return (
   539→    <div className="space-y-2">
   540→      {list.map((item, i) => (
   541→        <div key={`${item}-${i}`} className="flex items-center gap-2 rounded-lg border p-2.5">
   542→          <span className="tnum flex h-7 w-7 shrink-0 items-center justify-center rounded-full bg-muted text-xs font-semibold">
   543→            {i + 1}
   544→          </span>
   545→          <span className="flex-1 text-sm">{item}</span>
   546→          <div className="flex shrink-0 gap-1">
   547→            <Button type="button" variant="ghost" size="sm" onClick={() => move(i, -1)} disabled={i === 0} aria-label={`Monter « ${item} »`}>
   548→              ↑
   549→            </Button>
   550→            <Button type="button" variant="ghost" size="sm" onClick={() => move(i, 1)} disabled={i === list.length - 1} aria-label={`Descendre « ${item} »`}>
   551→              ↓
   552→            </Button>
   553→          </div>
   554→        </div>
   555→      ))}

   665→      const fb: Record<number, CorrectionFeedback> = {};
   666→      for (const it of res.items) {
   667→        if (it.lastAnswer != null) restored[it.questionIdx] = it.lastAnswer;
   668→        if (it.feedback) fb[it.questionIdx] = it.feedback;

   678→    }
   679→  }, [sessionId]);

   685→  const session = data?.session ?? null;

   704→  function goTo(i: number) {
   705→    setCurrent(Math.max(0, Math.min(i, Math.max(0, questions.length - 1))));

   707→    setSqlResult(null);

   710→  async function validateAnswer() {

   712→    setSaving(true);
   713→    try {
   714→      const res = await api.post<{ ok: boolean; graded: boolean; feedback: CorrectionFeedback | null }>(
   715→        `/api/sessions/${session.id}/answer`,
   716→        { questionIdx: current, value: drafts[current] ?? null }
   717→      );
   718→      if (res.graded && res.feedback) {
   719→        setFeedbacks((prev) => ({ ...prev, [current]: res.feedback as CorrectionFeedback }));
   720→        const v = res.feedback.verdict;

   722→          title: v === "correct" ? "Réponse correcte" : v === "partial" ? "Réponse partiellement correcte" : "Réponse incorrecte",
   723→          description: immediate ? "Vous pouvez améliorer votre réponse puis revalider." : undefined,
   724→        });
   725→      } else {
   726→        toast({ title: "Réponse enregistrée", description: "Elle sera corrigée à la fin de la session." });
   727→        if (!isLast) goTo(current + 1);
   728→      }
   729→    } catch (e) {
   730→      toast({ title: "Échec de l'enregistrement", description: e instanceof Error ? e.message : undefined, variant: "destructive" });
   731→    } finally {
   732→      setSaving(false);
   733→    }
   734→  }

   736→  async function askHint() {

   776→      const res = await api.post<SqlRunResponse>("/api/sql/run", {
   777→        sql: drafts[current],
   778→        setupSql: q.fixtureSql ?? undefined,
   779→      });

   782→      toast({ title: "Exécution impossible", description: e instanceof Error ? e.message : undefined, variant: "destructive" });
   783→    } finally {
   784→      setExecuting(false);
   785→    }

   805→      <section className="space-y-6" aria-label="Session en cours de chargement">

   822→            Réessayer

   835→  }
   836→
   837→  if (!q) {
   838→    return (
   839→      <section className="space-y-4" aria-label="Session vide">
   840→        <Alert>
   841→          <Info className="h-4 w-4" aria-hidden />

   852→  const draft = drafts[current] ?? defaultDraft(q);

   854→  const remainingHints = hintRemaining[current] ?? Math.max(0, (q.hints ?? []).length);

   858→  return (
   859→    <section className="space-y-6" aria-labelledby="session-title">

   862→        <div className="flex min-w-0 items-center gap-3">
   863→          <Button variant="ghost" size="icon" onClick={onExit} aria-label="Quitter la session">
   864→            <ArrowLeft className="h-4 w-4" aria-hidden />

   887→            {answeredCount}/{questions.length} répondu(s)

  1191→function formatAnswerValue(q: Question, value: unknown): string {
  1192→  if (value == null) return "(aucune réponse)";
  1193→  if (typeof value === "string") return value;
  1194→  if (typeof value === "number" || typeof value === "boolean") return String(value);
  1195→  if (Array.isArray(value)) return value.map((v, i) => `${i + 1}. ${String(v)}`).join("\n");
  1196→  if (typeof value === "object") {
  1197→    const obj = value as Record<string, unknown>;
  1198→    if (q.type === "matching") return Object.entries(obj).map(([k, v]) => `${k} → ${String(v)}`).join("\n");
  1199→    if (q.type === "error_identification")
  1200→      return `Ligne : ${String(obj.line ?? "?")}\nExplication : ${String(obj.explanation ?? "?")}`;
  1201→    return Object.entries(obj).map(([k, v]) => `${k} : ${String(v)}`).join("\n");
  1202→  }
  1203→  return String(value);
  1204→}
