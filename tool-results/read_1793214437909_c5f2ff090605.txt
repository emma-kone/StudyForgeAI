   176→const dateFormatter = new Intl.DateTimeFormat("fr-FR", { dateStyle: "medium", timeStyle: "short" });

   178→function formatDate(iso: string): string {
   179→  const date = new Date(iso);

   183→function formatSize(size: number): string {

   187→/** Analyse défensive d'un champ d'analyse encodé en JSON (tableau de chaînes). */
   188→function parseStringArray(raw: string | null): string[] {
   189→  if (!raw) return [];
   190→  try {
   191→    const parsed: unknown = JSON.parse(raw);
   192→    if (!Array.isArray(parsed)) return [];
   193→    return parsed.map((item) => String(item).trim()).filter((item) => item.length > 0);
   194→  } catch {
   195→    const trimmed = raw.trim();
   196→    return trimmed.length > 0 ? [trimmed] : [];
   197→  }
   198→}

   208→function statusMeta(status: DocStatus): {
   209→  label: string;
   210→  variant: "default" | "destructive" | "secondary" | "outline";
   211→} {
   212→  switch (status) {
   213→    case "READY":
   214→      return { label: "Prêt", variant: "default" };
   215→    case "ERROR":
   216→      return { label: "Erreur", variant: "destructive" };
   217→    case "PROCESSING":
   218→      return { label: "Traitement…", variant: "secondary" };
   219→    default:
   220→      return { label: "En attente", variant: "outline" };
   221→  }
   222→}

   224→function errorMessage(err: unknown, fallback: string): string {
   225→  return err instanceof Error && err.message ? err.message : fallback;
   226→}

   323→                <RotateCw className="h-4 w-4" aria-hidden />

   332→            <Button size="sm" variant="outline" onClick={() => onAnalyze(doc.id)} disabled={analyzing}>
   333→              {analyzing ? (
   334→                <Loader2 className="h-4 w-4 animate-spin" aria-hidden />
   335→              ) : (
   336→                <Sparkles className="h-4 w-4" aria-hidden />
   337→              )}
   338→              {analyzing ? "Analyse…" : "Analyser (IA)"}
   339→            </Button>
   340→          )}
   341→          <Button size="sm" variant="ghost" onClick={() => onOpen(doc.id)}>
   342→            Détails
   343→          </Button>
   344→          <Button
   345→            size="icon"
   346→            variant="ghost"
   347→            className="ml-auto size-8 text-muted-foreground hover:text-destructive"
   348→            aria-label="Supprimer"
   349→            onClick={() => onDelete(doc)}
   350→          >
   351→            <Trash2 className="h-4 w-4" aria-hidden />
   352→          </Button>
   353→        </div>
   354→      </CardContent>
   355→    </Card>
   356→  );
   357→}

   436→      setDocsError(errorMessage(err, "Impossible de charger les documents."));

   438→      setDocsLoading(false);

   481→        const failCount = res.results.length - okCount;

   489→            title: "Import impossible",

   495→            title: "Import partiel",

   503→          description: errorMessage(err, "L'import des documents a échoué."),

   508→        if (fileInputRef.current) fileInputRef.current.value = "";

   531→      setCreatingSubject(false);

   549→      await loadSubjects();
   550→    } catch (err: unknown) {
   551→      toast({
   552→        title: "Ajout impossible",
   553→        description: errorMessage(err, "L'ajout du chapitre a échoué."),

   555→      });
   556→    } finally {
   557→      setCreatingChapter(false);
   558→    }
   559→  }, [newChapterName, importSubjectId, creatingChapter, selectedImportSubject, loadSubjects, toast]);
