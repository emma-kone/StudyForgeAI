   148→   148→function hashStr(s: string): number {
   149→   149→  let h = 2166136261;
   150→   150→  for (let i = 0; i < s.length; i++) {
   151→   151→    h ^= s.charCodeAt(i);
   152→   152→    h = Math.imul(h, 16777619);
   153→   153→  }
   154→   154→  return h >>> 0;
   155→   155→}

   157→   157→/** Mélange déterministe (même graine → même ordre, stable entre rendus). */
   158→   158→function seedShuffle<T>(arr: T[], seed: number): T[] {
   159→   159→  const copy = [...arr];
   160→   160→  let state = seed || 1;
   161→   161→  const rand = () => {
   162→   162→    state |= 0;
   163→   163→    state = (state + 0x6d2b79f5) | 0;
   164→   164→    let t = Math.imul(state ^ (state >>> 15), 1 | state);
   165→   165→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
   166→   166→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   167→   167→  };
   168→   168→  for (let i = copy.length - 1; i > 0; i--) {
   169→   169→    const j = Math.floor(rand() * (i + 1));
   170→   170→    [copy[i], copy[j]] = [copy[j], copy[i]];
   171→   171→  }
   172→   172→  return copy;
   173→   173→}

   430→   430→function FillBlankInput({ q, value, onChange }: { q: Question; value: unknown; onChange: (v: string[]) => void }) {
   431→   431→  const answers = Array.isArray(value) ? (value as string[]) : [];

   434→   434→  const set = (i: number, v: string) => {
   435→   435→    const next = [...answers];
   436→   436→    while (next.length < blanks.length) next.push("");

   438→   438→    onChange(next);
   439→   439→  };

   441→   441→    <div className="space-y-3">

   444→   444→          <span key={i} className="contents">
   445→   445→            <span className="whitespace-pre-wrap">{seg}</span>
   446→   446→            {i < blanks.length && (
   447→   447→              <Input
   448→   448→                value={answers[i] ?? ""}
   449→   449→                onChange={(e) => set(i, e.target.value)}
   450→   450→                className="inline-block w-40 font-medium"
   451→   451→                aria-label={`Trou ${i + 1}`}

   454→   454→            )}
   455→   455→          </span>

   457→   457→      </div>

   459→   459→        <div className="space-y-2">
   460→   460→          {blanks.slice(Math.max(0, segments.length - 1)).map((_, j) => {
   461→   461→            const idx = Math.max(0, segments.length - 1) + j;
   462→   462→            return (
   463→   463→              <Input
   464→   464→                key={idx}
   465→   465→                value={answers[idx] ?? ""}

   468→   468→                aria-label={`Trou ${idx + 1}`}
   469→   469→                autoComplete="off"
